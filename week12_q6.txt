
import java.util.*;

public class Main {
public static void main(String[] args) {
Scanner sc = new Scanner(System.in);

int n = sc.nextInt();
int[] arr = new int[n];

for (int i = 0; i < n; i++) {
arr[i] = sc.nextInt();
}

int k = sc.nextInt();

if (k <= 0 || k > n) {
System.out.println(0);
sc.close();
return;
}

Map<Integer, Integer> freq = new HashMap<>();
int distinct = 0;
int maxDistinct = 0;

for (int i = 0; i < n; i++) {
// Add the new element to the window
int value = arr[i];
freq.put(value, freq.getOrDefault(value, 0) + 1);

if (freq.get(value) == 1) {
distinct++;
}

// Remove the element that leaves the window
if (i >= k) {
int removed = arr[i - k];
freq.put(removed, freq.get(removed) - 1);

if (freq.get(removed) == 0) {
freq.remove(removed);
distinct--;
}
}

// Update maximum for each complete window
if (i >= k - 1) {
maxDistinct = Math.max(maxDistinct, distinct);
}
}

System.out.println(maxDistinct);
sc.close();
}
}
